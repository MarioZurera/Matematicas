\newtheorem{corollary}[theorem]{\thmnamecorollary}
\theoremstyle{definition}
\newtheorem{definition}[theorem]{\thmnamedefinition}
\newtheorem{exercise}[theorem]{\thmnameexercise}
\newtheorem{example}[theorem]{\thmnameexample}
